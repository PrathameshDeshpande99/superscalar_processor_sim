addi x5, x0, 512     # x5 = 512
addi x5, x5, 512     # x5 = 2048 (Base address)
addi x6, x0, 20       # x6 = 20 (n)
addi x9, x6, -1       # x9 = 19 (n - 1 limit for outer loop)
addi x7, x0, 0        # x7 = 0 (i = 0)

OUTER_LOOP: bge x7, x9, END       # If i >= 19, exit
addi x8, x0, 0        # j = 0
sub x10, x9, x7       # x10 = (n - 1) - i (Inner loop limit)

INNER_LOOP: bge x8, x10, NEXT_OUTER # If j >= limit, go to next outer pass

    add x11, x8, x8       # x11 = 2 * j
    add x11, x11, x11     # x11 = 4 * j
    add x11, x11, x5      # x11 = Base + (4 * j) -> Address of arr[j]
    addi x12, x11, 4      # x12 = Address of arr[j+1]

    lw x13, 0(x11)        # x13 = arr[j]
    lw x14, 0(x12)        # x14 = arr[j+1]

    blt x13, x14, NO_SWAP
    beq x13, x14, NO_SWAP

    sw x14, 0(x11)        # Store arr[j+1] in arr[j] location
    sw x13, 0(x12)        # Store arr[j] in arr[j+1] location

NO_SWAP: addi x8, x8, 1        # j = j + 1
    j INNER_LOOP

NEXT_OUTER: addi x7, x7, 1        # i = i + 1
    j OUTER_LOOP

END:
